    23	@dataclass
    24	class CommandResult:
    25	    """What a command wants done. `text` is shown; the flags ask the session for
    26	    something only the session can do."""
    27	    text: str = ""
    28	    quit: bool = False
    29	    clear: bool = False
    30	    compact: bool = False
    31	    prompt: str = ""
    32	    """Run this as a normal turn after showing `text`.
    33	
    34	    Lets a command hand work to the model without knowing anything about the
    35	    loop — /review is a diff plus a request to read it, and writing that by
    36	    hand every time is exactly the friction that stops it happening."""

    39	@dataclass
    40	class Command:
    41	    name: str
    42	    summary: str                 # one line, shown by /help
    43	    run: "Callable[[str, Session], Awaitable[CommandResult]]"
    44	    aliases: tuple[str, ...] = ()
    45	
    46	
    47	REGISTRY: dict[str, Command] = {}
    48	
    49	
    50	def register(command: Command) -> Command:
    51	    """Add a command. Later registration wins for a name — the registry is
    52	    ordinary state, and a caller replacing a builtin has said what they meant."""
    53	    REGISTRY[command.name] = command
    54	    for alias in command.aliases:
    55	        REGISTRY[alias] = command
    56	    return command

    58	
    59	def resolve(line: str) -> "tuple[Command | None, str]":
    60	    """Split `/name rest` into its command and argument string."""
    61	    body = line[1:].strip()
    62	    if not body:
    63	        return None, ""
    64	    name, _, rest = body.partition(" ")
    65	    return REGISTRY.get(name.lower()), rest.strip()

    68	def command(name: str, summary: str, *aliases: str):
    69	    def wrap(fn):
    70	        register(Command(name=name, summary=summary, run=fn, aliases=aliases))
    71	        return fn
    72	    return wrap

   208	# ── Git: what the agent actually changed ─────────────────────────────────────
   209	# The single most common question after a turn is "what did it do to my repo",
   210	# and the honest answer is git's, not the agent's. These run through the Cell so
   211	# they see the same working directory the agent does — including an active
   212	# worktree, where the operator's own `git diff` in another terminal would show
   213	# nothing at all.
   214	
   215	
   216	async def _cell_git(session: "Session", command: str) -> tuple[str, bool]:
   217	    """(output, ok). Never raises: a REPL command must not end the session."""
   218	    if session.cell is None:
   219	        return "No Cell attached.", False
   220	    try:
   221	        res = await session.cell.run(command, timeout=30)
   222	    except Exception as e:  # noqa: BLE001
   223	        return f"{type(e).__name__}: {e}", False
   224	    body = (res.stdout or "").rstrip() or (res.stderr or "").rstrip()
   225	    return body, res.exit_code == 0

   392	def _clipboard_command() -> list[str] | None:
   393	    """The platform's "read stdin into the clipboard" command, if there is one.
   394	
   395	    Done with the tool the OS already ships rather than a dependency: pyperclip
   396	    would be a third package to install, on every platform, to move one string.
   397	    """
   398	    import shutil
   399	    import sys
   400	
   401	    if sys.platform == "win32":
   402	        return ["clip"]
   403	    if sys.platform == "darwin":
   404	        return ["pbcopy"]
   405	    if shutil.which("wl-copy"):
   406	        return ["wl-copy"]
   407	    if shutil.which("xclip"):
   408	        return ["xclip", "-selection", "clipboard"]
   409	    if shutil.which("xsel"):
   410	        return ["xsel", "--clipboard", "--input"]
   411	    return None

   414	def _last_assistant_text(session: "Session") -> str:
   415	    for message in reversed(session.messages):
   416	        if message.get("role") != "assistant":
   417	            continue
   418	        content = message.get("content")
   419	        if isinstance(content, str):
   420	            return content
   421	        if isinstance(content, list):
   422	            parts = [b.get("text", "") for b in content
   423	                     if isinstance(b, dict) and b.get("type") == "text"]
   424	            if any(parts):
   425	                return "\n".join(p for p in parts if p)
   426	    return ""

   501	async def _commit(args: str, session: "Session") -> CommandResult:
   502	    """Commits through the CELL, so the work is attributed to the agent that
   503	    did it — the Cell carries the agent's git identity (agents/config.py).
   504	    Running this on the host instead would record the operator as the author of
   505	    code they did not write."""
   506	    status, ok = await _cell_git(session, "git status --porcelain")
   507	    if not ok:

   509	    if not status.strip():

   511	
   512	    message = args.strip()
   513	    if not message:
   514	        changed = len(status.splitlines())
   515	        return CommandResult(
   516	            f"  {changed} file(s) changed. A commit needs a message:\n"
   517	            "      /commit <what changed and why>\n\n"
   518	            "  Ask the agent to write one with /review first if you want a hand.")
   519	
   520	    staged, ok = await _cell_git(session, "git add -A")
   521	    if not ok:
   522	        return CommandResult(f"  Could not stage: {staged}")
   523	
   524	    # -- to stop a message beginning with a dash being read as a flag, and
   525	    # single quotes doubled so the shell keeps the message intact.
   526	    safe = message.replace("'", "'\''")
   527	    out, ok = await _cell_git(session, f"git commit -m '{safe}'")
   528	    if not ok:
   529	        return CommandResult(f"  Commit failed:\n  {out}")
   530	
   531	    who, _ = await _cell_git(session, "git log -1 --format=%an")
   532	    return CommandResult(f"  {out}\n  Authored by {who.strip() or 'unknown'}.")
   533	
   534	
   535	@command("review", "have the agent read its own uncommitted changes")
   536	async def _review(args: str, session: "Session") -> CommandResult:
   537	    """The one review that reliably does not happen is the one requiring the
   538	    operator to write the request. This assembles it: the diff is already on
   539	    disk, so the command supplies the ask and the agent reads what it wrote."""
   540	    stat, ok = await _cell_git(session, "git diff --stat HEAD")
   541	    if not ok:
   542	        return CommandResult(f"  {stat}")
   543	    if not stat.strip():
   544	        return CommandResult("  Nothing to review — the working tree is clean.")
   545	
   546	    focus = args.strip()
   547	    ask = (
   548	        "Review the uncommitted changes in this repository. Run "
   549	        "`git diff HEAD` to read them, then report: anything that looks wrong "
   550	        "or unfinished, anything that changes behaviour beyond what was asked, "
   551	        "and anything missing a test. Be specific — name files and lines. If it "
   552	        "looks correct, say so plainly rather than inventing concerns."
   553	    )
   554	    if focus:
   555	        ask += f"\n\nPay particular attention to: {focus}"
   556	    return CommandResult(
   557	        "\n".join(f"  {ln}" for ln in stat.splitlines()) + "\n", prompt=ask)

   561	
   562	
   563	@command("sessions", "conversations you can resume in this workspace", "ls")
   564	async def _sessions(args: str, session: "Session") -> CommandResult:
   565	    from forge.tui import persistence
   566	
   567	    entries = persistence.listing(session.workspace)
   568	    if not entries:
   569	        return CommandResult(
   570	            "  No saved conversations here yet. They are written after each\n"
   571	            "  turn, into .forge/sessions/.")
   572	    lines = []
   573	    for i, e in enumerate(entries, 1):
   574	        current = "  (this one)" if e.id == session.session_id else ""
   575	        lines.append(f"  {i:>2}. {e.title}")
   576	        lines.append(f"      {e.agent_id} · {e.turns} turns · {e.age} · {e.id}{current}")
   577	    lines.append("")
   578	    lines.append("  /resume <number|id> to pick one up.")
   579	    return CommandResult("\n".join(lines))

   581	
   582	@command("resume", "continue an earlier conversation")
   583	async def _resume(args: str, session: "Session") -> CommandResult:
   584	    """Replaces the live transcript with a stored one.
   585	
   586	    The Cell, the tools and the allowlist are NOT restored — they are rebuilt
   587	    from the current profile and the current workspace, which is the honest
   588	    thing to do: a conversation from last week describes files as they were,
   589	    and pretending otherwise would let the model edit against remembered text.
   590	    Read-tracking starts empty for the same reason, so the first edit after a
   591	    resume has to read the file again.
   592	    """
   593	    from forge.tui import persistence
   594	
   595	    entries = persistence.listing(session.workspace)
   596	    if not entries:
   597	        return CommandResult("  Nothing to resume in this workspace.")
   598	
   599	    picked = persistence.resolve(session.workspace, args)
   600	    if picked is None:
   601	        return CommandResult(
   602	            f"  No session matches {args.strip()!r}. Try /sessions for the list.")
   603	    if picked.id == session.session_id:
   604	        return CommandResult("  That is the conversation you are already in.")
   605	
   606	    messages = persistence.load(session.workspace, picked.id)
   607	    if messages is None:
   608	        return CommandResult(f"  Could not read session {picked.id}.")
   609	
   610	    session.messages = messages
   611	    session.session_id = picked.id
   612	    session.turns = picked.turns
   613	    session.last_truncated = None
   614	    return CommandResult(
   615	        f"  Resumed: {picked.title}\n"
   616	        f"  {picked.turns} turns, {len(messages)} messages, last touched {picked.age}.\n"
   617	        "  Files may have changed since — the agent will re-read before editing.")

   620	# ── Project conventions ──────────────────────────────────────────────────────
   621	
   622	
   623	@command("init", "write an AGENTS.md so the agent knows this project's conventions")
   624	async def _init(args: str, session: "Session") -> CommandResult:
   625	    """Asks the agent to read the repository and write down its conventions.
   626	
   627	    Without one, an agent rediscovers a project by being corrected: wrong test
   628	    runner, wrong docstring style, a dependency the project deliberately
   629	    avoids. Each costs a turn to hit and another to fix, every session.
   630	
   631	    The agent writes it rather than a template, because a useful conventions
   632	    file is specific to what is actually in the repository, and a generic one
   633	    is worse than none — it occupies the same space on every turn while saying
   634	    nothing.
   635	    """
   636	    from forge.agents import conventions
   637	
   638	    existing = conventions.find(session.workspace)
   639	    if existing is not None and args.strip().lower() != "force":
   640	        rel = existing.relative_to(session.workspace)
   641	        return CommandResult(
   642	            f"  {rel} already exists and is loaded into every turn.\n"
   643	            "  Edit it directly, or /init force to have the agent rewrite it.")
   644	
   645	    target = conventions.FILENAMES[0]
   646	    return CommandResult(
   647	        f"  Asking the agent to survey this repository and write {target}.\n",
   648	        prompt=(
   649	            f"Write {target} for this repository: the conventions someone "
   650	            "working here has to know, which are not obvious from a single "
   651	            "file.\n\n"
   652	            "First look: the build and dependency files, the test setup, the "
   653	            "directory layout, and a few representative source files. Use "
   654	            "glob, grep and read_file — do not guess from the directory names.\n\n"
   655	            "Then write it, covering only what you actually verified:\n"
   656	            "- what this project is, in two sentences\n"
   657	            "- how to install, run and test it, with the real commands\n"
   658	            "- the layout, and where a new module of each kind belongs\n"
   659	            "- conventions visible in the code: naming, error handling, typing, "
   660	            "test style, anything the project does consistently and unusually\n"
   661	            "- anything deliberately avoided, where the code shows it\n\n"
   662	            "Be specific and short. This file is sent to the model on every "
   663	            "turn, so a sentence that does not change what an agent would do is "
   664	            "costing money to say nothing. Do not include a licence, a feature "
   665	            "list, or anything you could have written without reading the repo."
   666	        ))
